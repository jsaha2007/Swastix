DOCTORS = [
    {"id": 1, "firstName": "John", "lastName": "Doe"},
    {"id": 2, "firstName": "Alice", "lastName": "Smith"},
    {"id": 3, "firstName": "Bob", "lastName": "Brown"},
    {"id": 4, "firstName": "Maria", "lastName": "García"},
    {"id": 5, "firstName": "Liu", "lastName": "Wei"},
]

def get_doctor_by_id(doctor_id):
    """
    Retrieve a doctor's basic profile (firstName, lastName) by ID.
    Returns the doctor dict if found, otherwise returns None.
    """
    for doctor in DOCTORS:
        if doctor["id"] == doctor_id:
            return {"firstName": doctor["firstName"], "lastName": doctor["lastName"]}
    return None
