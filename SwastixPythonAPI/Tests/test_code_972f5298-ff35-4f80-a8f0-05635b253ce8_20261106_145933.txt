# test_db.py
import builtins
import types
import pytest
from app import db as app_db


class DummySession:
    def __init__(self):
        self.closed = False

    def close(self):
        self.closed = True


def test_get_db_yields_session_and_closes(monkeypatch):
    dummy = DummySession()
    # Replace SessionLocal with a callable that returns dummy session
    monkeypatch.setattr(app_db, "SessionLocal", lambda: dummy)

    gen = app_db.get_db()
    # First yield should give us the dummy session
    session = next(gen)
    assert session is dummy
    # After generator is exhausted, the finally block should close the session
    with pytest.raises(StopIteration):
        next(gen)
    assert dummy.closed is True


# test_doctor_repository.py
import pytest
from sqlalchemy import create_engine
from sqlalchemy.orm import sessionmaker
from app.db import Base
from app.models.doctor import Doctor
from app.repositories.doctor_repository import DoctorRepository


@pytest.fixture
def db_session():
    engine = create_engine("sqlite:///:memory:", connect_args={"check_same_thread": False})
    Base.metadata.create_all(bind=engine)
    SessionLocal = sessionmaker(bind=engine, autocommit=False, autoflush=False)
    session = SessionLocal()
    yield session
    session.close()
    engine.dispose()


def test_get_by_id_returns_doctor(db_session):
    doctor = Doctor(first_name="John", last_name="Doe", specialty="Cardiology", email="john@example.com")
    db_session.add(doctor)
    db_session.commit()
    repo = DoctorRepository(db_session)
    fetched = repo.get_by_id(doctor.id)
    assert fetched is not None
    assert fetched.first_name == "John"
    assert fetched.last_name == "Doe"


def test_get_by_id_returns_none_when_not_found(db_session):
    repo = DoctorRepository(db_session)
    assert repo.get_by_id(999) is None

# test_auth.py
import pytest
from fastapi import HTTPException, status
from app.dependencies import auth


def test_admin_required_allows_admin():
    # get_current_user_role returns "admin" by default
    result = auth.admin_required()
    assert result is True


def test_admin_required_denies_non_admin(monkeypatch):
    # Override get_current_user_role to return "user"
    monkeypatch.setattr(auth, "get_current_user_role", lambda: "user")
    with pytest.raises(HTTPException) as exc:
        auth.admin_required()
    assert exc.value.status_code == status.HTTP_403_FORBIDDEN
    assert exc.value.detail == "Insufficient privileges"

@pytest.fixture
def client(db_session):
    # Override the get_db dependency to use the test session
    def get_test_db():
        try:
            yield db_session
        finally:
            pass

    app.dependency_overrides[get_db] = get_test_db
    with TestClient(app) as c:
        yield c
    app.dependency_overrides.clear()


def test_get_doctor_basic_profile_success(client, db_session):
    doctor = Doctor(first_name="Maria", last_name="García")
    db_session.add(doctor)
    db_session.commit()

def test_get_doctor_basic_profile_not_found(client):
